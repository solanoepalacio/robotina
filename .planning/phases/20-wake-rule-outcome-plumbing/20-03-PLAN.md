```python
def _check_and_dispatch_wake(
    parent_invocation_id: str | None,
    session: Session,
    queue=None,
) -> None:
    """Idempotently dispatch a follow-up RobotinaInvocation when all workflows
    sharing parent_invocation_id reach terminal status.

    MUST be called inside the caller's session — does NOT commit. The caller
    performs the final commit so the terminal-status write and the new
    invocation row land atomically (PITFALLS.md Pitfall 2).

    Idempotency (PITFALLS.md Pitfall 1): uses an atomic UPDATE ... WHERE
    wake_dispatched_at IS NULL RETURNING id guard. If 0 rows are affected,
    the wake already fired and we skip.

    Crash safety (PITFALLS.md Pitfall 11): pre-assigns rq_job_id BEFORE
    commit so the startup reconciler can detect orphans where the commit
    succeeded but enqueue did not.

    If queue is None, the DB-side state still mutates (wake_dispatched_at
    flips, new invocation row inserted) but no enqueue happens. The
    reconciler picks up on next boot.
    """
    import uuid
    from datetime import datetime
    from robotina.queue.models import (
        RobotinaInvocation, InvocationTrigger, InvocationStatus,
        WorkflowRun, WorkflowStatus,
    )

    if parent_invocation_id is None:
        return  # Pre-Phase-18 row with no invocation FK — no wake to dispatch

    # All-terminal check
    total = session.query(WorkflowRun).filter(
        WorkflowRun.triggered_by_invocation_id == parent_invocation_id
    ).count()
    if total == 0:
        return
    terminal = session.query(WorkflowRun).filter(
        WorkflowRun.triggered_by_invocation_id == parent_invocation_id,
        WorkflowRun.status.in_([WorkflowStatus.DONE, WorkflowStatus.FAILED]),
    ).count()
    if terminal != total:
        return  # Not all workflows terminal yet

    # Idempotency guard — atomic UPDATE ... WHERE wake_dispatched_at IS NULL
    updated = session.execute(
        text(
            "UPDATE robotina_invocations "
            "SET wake_dispatched_at = :now "
            "WHERE id = :id AND wake_dispatched_at IS NULL "
            "RETURNING id"
        ),
        {"id": parent_invocation_id, "now": datetime.utcnow()},
    ).fetchone()
    if updated is None:
        logger.info("wake already dispatched | parent_invocation_id=%s", parent_invocation_id)
        return

    parent = session.get(RobotinaInvocation, parent_invocation_id)

    # Pre-assign rq_job_id (D-07 / Pitfall 11)
    pre_assigned_job_id = str(uuid.uuid4())
    new_inv = RobotinaInvocation(
        id=str(uuid.uuid4()),
        conversation_id=parent.conversation_id,
        trigger=InvocationTrigger.WORKFLOW_COMPLETION,
        trigger_ref_id=parent.id,
        rq_job_id=pre_assigned_job_id,
        status=InvocationStatus.PENDING,
    )
    session.add(new_inv)
    session.flush()  # materialize new_inv.id

    # Build the WakeInvocationInput payload from the linked workflow outcomes
    from robotina.queue.task_types import (
        WakeInvocationInput, WorkflowOutcomeSummary, AddRecipeOutcome,
    )
    runs = session.query(WorkflowRun).filter(
        WorkflowRun.triggered_by_invocation_id == parent_invocation_id
    ).all()
    summaries = [
        WorkflowOutcomeSummary(
            workflow_run_id=r.id,
            workflow_type=r.workflow_type,
            status="done" if r.status == WorkflowStatus.DONE else "failed",
            outcome=AddRecipeOutcome(**r.outcome) if r.outcome else None,
        )
        for r in runs
    ]
    wake_input = WakeInvocationInput(
        previous_invocation_id=parent.id,
        conversation_id=parent.conversation_id,
        outcomes=summaries,
    )

    # Enqueue (skip if queue is None — reconciler will recover)
    if queue is not None:
        try:
            queue.enqueue(
                "robotina.queue.jobs.run_task",
                wake_input,
                job_id=pre_assigned_job_id,
                meta={"task_type": "handle-incoming-message", "invocation_id": new_inv.id},
                result_ttl=-1,
                failure_ttl=-1,
            )
        except Exception:
            logger.exception(
                "wake enqueue failed | new_invocation_id=%s job_id=%s",
                new_inv.id, pre_assigned_job_id,
            )
            raise

    logger.info(
        "wake dispatched | parent_invocation_id=%s new_invocation_id=%s rq_job_id=%s outcomes=%d",
        parent_invocation_id, new_inv.id, pre_assigned_job_id, len(summaries),
    )
```

if task_type == "handle-incoming-message":
    from datetime import datetime
    from robotina.queue.models import RobotinaInvocation, InvocationStatus, InvocationTrigger
    from robotina.queue.task_types import WakeInvocationInput, IncomingMessageInput
    from robotina.db import SessionLocal

    job = get_current_job()
    invocation_id = job.meta["invocation_id"]

    with SessionLocal() as session:
        inv = session.get(RobotinaInvocation, invocation_id)
        if inv is None:
            raise RuntimeError(f"RobotinaInvocation {invocation_id} not found")

        # PENDING → RUNNING
        inv.status = InvocationStatus.RUNNING
        inv.started_at = datetime.utcnow()
        session.commit()

        try:
            if inv.trigger == InvocationTrigger.USER_MESSAGE:
                result = _run_user_message_turn(inv, task_input, session)
            elif inv.trigger == InvocationTrigger.WORKFLOW_COMPLETION:
                if not isinstance(task_input, WakeInvocationInput):
                    task_input = WakeInvocationInput(**task_input) if isinstance(task_input, dict) else task_input
                result = _run_wake_turn(inv, task_input, session)
            else:
                raise RuntimeError(f"unsupported invocation trigger {inv.trigger}")
            inv.status = InvocationStatus.DONE
        except Exception:
            inv.status = InvocationStatus.FAILED
            inv.completed_at = datetime.utcnow()
            session.commit()
            raise
        finally:
            if inv.status != InvocationStatus.FAILED:
                inv.completed_at = datetime.utcnow()
                session.commit()

        return result
```

def _render_wake_user_message(wake_input: "WakeInvocationInput") -> str:
    lines = ["Los siguientes flujos terminaron:"]
    for s in wake_input.outcomes:
        if s.status == "done" and s.outcome and s.outcome.status == "success":
            lines.append(f"- ✓ {s.outcome.recipe_name} (id: {s.outcome.recipe_id})")
        elif s.status == "failed":
            reason = s.outcome.failure_reason if s.outcome else "(sin detalle)"
            lines.append(f"- ✗ Falló: {reason}")
        else:
            lines.append(f"- ? {s.workflow_type} {s.status}")
    lines.append("(Wake-trigger; el usuario ya fue notificado.)")
    return "\n".join(lines)
```

## Handling wake-context invocations

When the user message begins with `Los siguientes flujos terminaron:` and ends with
`(Wake-trigger; el usuario ya fue notificado.)`, you are being woken to reflect on
completed workflows from a previous turn — NOT to respond to a new user request.

In this wake-context mode:

  reasoning, which is captured in logs and traces) and then terminate cleanly
  by producing no further tool calls. Do NOT call start-workflow unless the
  outcomes reveal an action you should take (typically: nothing).
- If outcomes include failures with unclear reasons, note them in your reasoning
  for the operator (visible in LangWatch traces) but do not act on them — the
  user has already been informed.

End your wake-context turns by producing a final assistant message that is a
single short Spanish acknowledgement (e.g. "Vistos los resultados.") — this is
captured in logs but does not reach the user.
