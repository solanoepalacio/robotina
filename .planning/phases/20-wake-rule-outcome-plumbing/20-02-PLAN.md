# Phase 20 / WAKE-04 — WakeInvocationInput and helpers
#
# Wake-context input dispatched by run_task when an invocation has
# trigger=WORKFLOW_COMPLETION. Built by _check_and_dispatch_wake from the
# WorkflowRuns linked to the parent invocation.

class WorkflowOutcomeSummary(BaseModel):
    """Thin envelope around a single workflow's outcome for the wake input.

    Phase 20 (WAKE-04): one summary per WorkflowRun whose
    triggered_by_invocation_id == parent_invocation.id.

    `outcome` is populated for DONE workflows (composed by finalize-outcome)
    and None for FAILED workflows (no finalize-outcome ran). The agent uses
    `status` to know which is which.
    """
    model_config = ConfigDict(extra="forbid")
    workflow_run_id: str
    workflow_type: str
    status: Literal["done", "failed"]
    outcome: AddRecipeOutcome | None = None


class WakeInvocationInput(BaseModel):
    """Input model for wake-context Robotina invocations (trigger=workflow_completion)."""
    model_config = ConfigDict(extra="forbid")
    previous_invocation_id: str
    conversation_id: str
    outcomes: list[WorkflowOutcomeSummary]


class FinalizeOutcomeInput(BaseModel):
    """Input for the deterministic finalize-outcome task (Phase 20 D-03).

    Composes an AddRecipeOutcome from accumulated workflow artifacts.
    """
    model_config = ConfigDict(extra="forbid")
    metadata: dict | None = None
    load: dict | None = None
    failure_reason: str | None = None
```

elif task_type == "finalize-outcome":
    # Phase 20 / D-01: deterministic outcome composition
    from robotina.queue.task_types import FinalizeOutcomeInput, AddRecipeOutcome
    from robotina.db import SessionLocal
    from robotina.queue.models import WorkflowRun, WorkflowRunStep

    fi = task_input if isinstance(task_input, FinalizeOutcomeInput) else FinalizeOutcomeInput(**task_input)

    # Compose outcome from accumulated artifacts
    if fi.load and fi.load.get("recipe_id"):
        outcome = AddRecipeOutcome(
            status="success",
            recipe_id=fi.load.get("recipe_id"),
            recipe_name=(fi.metadata or {}).get("recipe_name") or fi.load.get("recipe_name"),
            recipe_slug=(fi.metadata or {}).get("recipe_slug") or fi.load.get("recipe_slug"),
            image_present=False,  # Phase 24 fills the True case
        )
    else:
        outcome = AddRecipeOutcome(
            status="failure",
            failure_reason=fi.failure_reason or "finalize-outcome called without a load artifact",
        )

    # Write to WorkflowRun.outcome — the row is identified by the current step's workflow_run_id
    with SessionLocal() as session:
        job = get_current_job()
        step = session.query(WorkflowRunStep).filter_by(task_job_id=job.id).one()
        run = session.get(WorkflowRun, step.workflow_run_id)
        run.outcome = outcome.model_dump()
        session.commit()

    return outcome.model_dump()
```

Note: this branch runs BEFORE the `handle-incoming-message` branch reads `job.meta["invocation_id"]`, so it's an isolated deterministic path.
