```python
"""Startup reconciler for orphan RobotinaInvocations (WAKE-05).

After a worker crash between the wake-helper's commit and the RQ enqueue,
a RobotinaInvocation row exists with status=PENDING, wake_dispatched_at set,
and rq_job_id set — but no corresponding RQ job exists. AOF persisted the
commit but cannot replay the enqueue (it's RQ-side).

This module's reconcile_invocations() is called at task-runner boot to
detect these orphans and re-enqueue with the pre-assigned job_id. RQ
deduplicates on job_id so a race-win re-enqueue is safe even if the
original eventually showed up.

Out of scope: WorkflowRunStep orphans (Pitfall 11 freebie). Deferred to
a backlog phase.
"""

import logging
from typing import TYPE_CHECKING

from sqlalchemy.orm import Session

from robotina.queue.models import RobotinaInvocation, InvocationStatus

if TYPE_CHECKING:
    from rq import Queue

logger = logging.getLogger(__name__)


def reconcile_invocations(session: Session, queue: "Queue") -> int:
    """Scan for orphan wake invocations and re-enqueue them.

    Returns the count of invocations re-enqueued.
    """
    from rq.job import Job

    candidates = (
        session.query(RobotinaInvocation)
        .filter(
            RobotinaInvocation.status == InvocationStatus.PENDING,
            RobotinaInvocation.wake_dispatched_at.isnot(None),
            RobotinaInvocation.rq_job_id.isnot(None),
        )
        .all()
    )

    reconciled = 0
    for inv in candidates:
        try:
            if Job.exists(inv.rq_job_id, connection=queue.connection):
                logger.debug("invocation %s healthy (rq job %s present)", inv.id, inv.rq_job_id)
                continue
        except Exception:
            logger.exception("Job.exists check failed for invocation %s", inv.id)
            continue

        # Orphan — re-enqueue with the same pre-assigned job_id
        try:
            from robotina.queue.task_types import WakeInvocationInput, WorkflowOutcomeSummary, AddRecipeOutcome
            from robotina.queue.models import WorkflowRun, WorkflowStatus

            parent_id = inv.trigger_ref_id
            runs = session.query(WorkflowRun).filter(
                WorkflowRun.triggered_by_invocation_id == parent_id
            ).all()
            summaries = [
                WorkflowOutcomeSummary(
                    workflow_run_id=r.id,
                    workflow_type=r.workflow_type,
                    status="done" if r.status == WorkflowStatus.DONE else "failed",
                    outcome=AddRecipeOutcome(**r.outcome) if r.outcome else None,
                )
                for r in runs
            ]
            wake_input = WakeInvocationInput(
                previous_invocation_id=parent_id,
                conversation_id=inv.conversation_id,
                outcomes=summaries,
            )

            queue.enqueue(
                "robotina.queue.jobs.run_task",
                wake_input,
                job_id=inv.rq_job_id,
                meta={"task_type": "handle-incoming-message", "invocation_id": inv.id},
                result_ttl=-1,
                failure_ttl=-1,
            )
            reconciled += 1
            logger.info(
                "reconciled orphan invocation | id=%s rq_job_id=%s",
                inv.id, inv.rq_job_id,
            )
        except Exception:
            logger.exception("failed to reconcile invocation %s", inv.id)

    if reconciled > 0:
        logger.info("reconcile_invocations: re-enqueued %d orphan invocation(s)", reconciled)
    return reconciled
```
