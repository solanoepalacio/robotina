    ```python
    # src/robotina/agent/middleware.py
    """LangChain agent middleware for Robotina structured-action logging.

    Phase 12 (OBS-06) replacement for the legacy
    ``robotina.agent.callbacks.AgentLoggingHandler``. LangWatch tracing remains
    on the LangChain callback bus (``langwatch.langchain.LangChainTracer`` in
    ``robotina.queue.jobs``) — only the per-agent log lines move to middleware.
    """
    from __future__ import annotations

    import logging
    from typing import Callable

    from langchain.agents.middleware import (
        after_model,
        wrap_model_call,
        wrap_tool_call,
        AgentState,
        ModelRequest,
        ModelResponse,
        ToolCallRequest,
    )  # AGENT-13 / Phase 12
    from langchain_core.messages import AIMessage, ToolMessage
    from langgraph.runtime import Runtime
    from langgraph.types import Command

    logger = logging.getLogger(__name__)


    @wrap_model_call
    def log_around_model_call(
        request: ModelRequest,
        handler: Callable[[ModelRequest], ModelResponse],
    ) -> ModelResponse:
        """Log ``LLM stream start | model=<ChatClassName>`` before each model call.

        Replaces AgentLoggingHandler.on_chat_model_start. See PATTERNS.md
        §"Log-line emission pattern" for the byte-for-byte format contract.
        """
        model_name = type(request.model).__name__
        logger.info("LLM stream start | model=%s", model_name)
        return handler(request)


    @after_model
    def log_after_model(state: AgentState, runtime: Runtime) -> None:
        """Log ``Thinking | <reasoning>`` when the latest AI message carries
        ``additional_kwargs.reasoning_content`` (Ollama / Anthropic populate this;
        OpenAI does not — same behavior as the legacy callback, no regression).
        """
        messages = state.get("messages") or []
        if not messages:
            return
        last = messages[-1]
        if not isinstance(last, AIMessage):
            return
        thinking = last.additional_kwargs.get("reasoning_content")
        if thinking:
            logger.info("Thinking | %s", thinking)


    @wrap_tool_call
    def log_wrap_tool_call(
        request: ToolCallRequest,
        handler: Callable[[ToolCallRequest], ToolMessage | Command],
    ) -> ToolMessage | Command:
        """Log ``Tool call | tool=... input=...`` then ``Tool result | output=...``.

        Both input and output truncated to 200 chars — preserves the V5/V7
        security invariant from the legacy callback (see RESEARCH.md §Security
        Domain). Handler is invoked EXACTLY ONCE per call (Pitfall 2).
        """
        name = request.tool_call.get("name")
        args = request.tool_call.get("args", {})
        logger.info("Tool call | tool=%s input=%s", name, str(args)[:200])

        result = handler(request)

        if isinstance(result, ToolMessage):
            logger.info("Tool result | output=%s", str(result.content)[:200])
        else:
            logger.info("Tool result | output=<Command>")
        return result
    ```

<threat_model>

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Tool args → log sink | Untrusted user input flows into tool args (e.g., `endpoint`, query strings), then into `logger.info("Tool call | ... input=%s", str(args)[:200])`. The 200-char truncation is the existing mitigation, carried forward from the legacy callback. |
| Model output → log sink | Tool results (`ToolMessage.content`) and AI reasoning content (`additional_kwargs.reasoning_content`) are user-influence-derived strings that get logged. Same 200-char truncation applies to tool results; reasoning content is untruncated (matches legacy behavior). |
| Internal module import | `from robotina.agent.middleware import ...` runs at process startup. The module body must not perform I/O, env reads, or network calls — middleware instances are module-level singletons shared across all jobs (RESEARCH.md Pitfall 4). |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-12-01 | Information disclosure | `log_wrap_tool_call` logging untrusted tool args (V5/V7 ASVS, log injection / log bloat) | mitigate | Preserve 200-char truncation byte-for-byte from legacy callback: `str(args)[:200]` and `str(result.content)[:200]`. Regression test `test_log_wrap_tool_call_truncates_output_to_200_chars` in Task 1.1 asserts this invariant directly. |
| T-12-02 | Tampering / Information disclosure | Module-level middleware singletons leaking per-job data via closures (e.g., chat_id) | mitigate | RESEARCH.md Pitfall 4 — middleware module is stateless. No closures over per-job data. Per-job context flows through `state` / `runtime`, never through capture. Code review rule documented in middleware module docstring; covered structurally by the fact that the three middleware functions take only `(state, runtime)` or `(request, handler)` and reference no module-level mutable state. |
| T-12-03 | Tampering | Double-execution of side-effecting tool calls in `@wrap_tool_call` (e.g., `HouseholdManagerApiTool` POSTs twice) | mitigate | RESEARCH.md Pitfall 2 — exactly one `handler(request)` call per code path. Regression test `test_log_wrap_tool_call_invokes_handler_once` asserts handler.call_count == 1. |
| T-12-04 | Repudiation | Loss of LangWatch trace fidelity due to mis-wiring (e.g., removing the LangChainTracer callback) | accept (this plan) / mitigate (Plan 12-02) | This plan does NOT modify the LangWatch callback wiring in `jobs.py` — coexistence by design. Plan 12-02's threat model covers the diff that removes `AgentLoggingHandler` while keeping `langwatch.langchain.LangChainTracer`. |
| T-12-05 | Denial of service | Middleware import performs I/O / heavy work at process startup → worker boot time degrades | accept (low risk) | The module imports only LangChain types + stdlib logging. Verified by the existing `test_no_llm_instantiation_at_import_time` (test_agent_runner.py:90-149) which patches Chat* classes to raise during import — middleware module respects the same discipline. |
